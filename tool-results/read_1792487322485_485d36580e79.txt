    37→/** کاربر جاری از نشست — با بررسی فعال بودن حساب */
    38→export async function currentAuthUser(): Promise<AuthUser | null> {
    39→  const session = getSession();
    40→  if (!session) return null;
    41→  try {
    42→    const users = await load<UserRec[]>('users');
    43→    const u = users.find((x) => x.id === session.uid);
    44→    if (!u || !u.active) return null;
    45→    return {
    46→      id: u.id, username: u.username, name: u.name, role: u.role, active: u.active,
    47→      phone: u.phone ?? null, teacherId: u.teacherId ?? null, studentId: u.studentId ?? null,
    48→    };
    49→  } catch {
    50→    return null;
    51→  }
    52→}

    61→export async function authPost(body: Record<string, unknown>) {
    62→  const action = str(body.action);
    63→
    64→  if (action === 'login') {
    65→    const username = str(body.username).toLowerCase();
    66→    const password = str(body.password);
    67→    if (!username || !password) throw new ApiError('نام کاربری و رمز عبور را وارد کنید');
    68→    const users = await load<UserRec[]>('users', { maxAgeMs: 120_000 });
    69→    const user = users.find((u) => (u.username || '').toLowerCase() === username);
    70→    if (!user || !(await verifyPassword(password, user.password || ''))) {
    71→      throw new ApiError('نام کاربری یا رمز عبور نادرست است', 401);
    72→    }
    73→    if (!user.active) {
    74→      throw new ApiError('حساب کاربری شما توسط مدیر غیرفعال شده است. لطفاً با مدیر مکتب تماس بگیرید.', 403);
    75→    }
    76→    setSession(user.id);
    77→    return { user: toSessionUser(user) };
    78→  }
    79→
    80→  if (action === 'logout') {
    81→    clearSession();
    82→    return { ok: true };
    83→  }
    84→
    85→  if (action === 'change-password') {
    86→    const me = requireUser(await currentAuthUser());
    87→    const current = str(body.currentPassword);
    88→    const next = str(body.newPassword);
    89→    if (next.length < 6) throw new ApiError('رمز عبور جدید باید حداقل ۶ حرف باشد');
    90→    await mutate('users', async (users) => {
    91→      const u = users.find((x) => x.id === me.id) as UserRec | undefined;
    92→      if (!u || !(await verifyPassword(current, u.password || ''))) {
    93→        throw new ApiError('رمز عبور فعلی نادرست است');
    94→      }
    95→      u.password = await hashPassword(next);
    96→      return { ok: true };
    97→    }, `users: تغییر رمز عبور ${me.username}`);
    98→    return { ok: true };
    99→  }
   100→
   101→  throw new ApiError('عملیات نامعلوم');
   102→}

   202→export async function studentsGet(me: AuthUser | null, sp: URLSearchParams) {
   203→  const user = requireUser(me);
   204→  const classId = sp.get('classId');
   205→  const search = sp.get('q');
   206→
   207→  if (user.role === 'STUDENT' || user.role === 'GUARDIAN') {
   208→    if (!user.studentId) throw new ApiError('حساب شما به هیچ شاگردی پیوند ندارد', 403);
   209→    const [students, classes] = await Promise.all([load<StudentRec[]>('students'), load<Rec[]>('classes', { maxAgeMs: 300_000 })]);
   210→    const s = students.find((x) => x.id === user.studentId);
   211→    const rows = s ? [{ ...s, className: classes.find((c) => c.id === s.classId)?.name || null }] : [];
   212→    return { students: rows };
   213→  }
   214→
   215→  const [students, classes] = await Promise.all([
   216→    load<StudentRec[]>('students'),
   217→    load<Rec[]>('classes', { maxAgeMs: 300_000 }),
   218→  ]);
   219→  let rows = students;
   220→  if (classId) rows = rows.filter((s) => s.classId === classId);
   221→  if (search) {
   222→    const q = search.toLowerCase();
   223→    rows = rows.filter((s) =>
   224→      [s.firstName, s.lastName, s.fatherName].some((f) => (f || '').toLowerCase().includes(q)));
   225→  }
   226→  rows = [...rows].sort((a, b) =>
   227→    (a.classId || '').localeCompare(b.classId || '') ||
   228→    ((a.rollNo ?? 1e9) - (b.rollNo ?? 1e9)) ||
   229→    (a.createdAt || '').localeCompare(b.createdAt || ''));
   230→  return {
   231→    students: rows.map((s) => ({ ...s, className: classes.find((c) => c.id === s.classId)?.name || null })),
   232→  };
   233→}

   417→interface ClassRec extends Rec {

   424→export async function classesGet(me: AuthUser | null) {
   425→  requireUser(me);
   426→  const [classes, classSubjects, subjects, teachers, students] = await Promise.all([
   427→    load<ClassRec[]>('classes'),
   428→    load<ClassSubjectRec[]>('classSubjects'),
   429→    load<Rec[]>('subjects', { maxAgeMs: 300_000 }),
   430→    load<TeacherRec[]>('teachers', { maxAgeMs: 120_000 }),
   431→    load<StudentRec[]>('students', { maxAgeMs: 120_000 }),
   432→  ]);
   433→  const sorted = [...classes].sort((a, b) => a.grade - b.grade);
   434→  return {
   435→    classes: sorted.map((c) => {
   436→      const supervisor = teachers.find((t) => t.id === c.supervisorId);
   437→      const csList = classSubjects
   438→        .filter((cs) => cs.classId === c.id)
   439→        .sort((a, b) => (a.createdAt || '').localeCompare(b.createdAt || ''))
   440→        .map((cs) => {
   441→          const subject = subjects.find((s) => s.id === cs.subjectId);
   442→          const teacher = teachers.find((t) => t.id === cs.teacherId);
   443→          return {

   449→      return {

   508→export async function assignPost(me: AuthUser | null, body: Record<string, unknown>) {
   509→  requireRole(me, 'ADMIN', 'HEADTEACHER');
   510→  const classId = str(body.classId);
   511→  const subjectId = str(body.subjectId);
   512→  const teacherId = str(body.teacherId) || null;
   513→  if (!classId || !subjectId) throw new ApiError('صنف و مضمون لازم است');
   514→  const [classes, subjects] = await Promise.all([
   515→    load<ClassRec[]>('classes'), load<Rec[]>('subjects', { maxAgeMs: 300_000 }),
   516→  ]);
   517→  if (!classes.find((c) => c.id === classId) || !subjects.find((s) => s.id === subjectId)) {
   518→    throw new ApiError('صنف یا مضمون یافت نشد');
   519→  }
   520→  await mutate('classSubjects', (cs) => {
   521→    const existing = cs.find((x) => x.classId === classId && x.subjectId === subjectId);
   522→    if (existing) {
   523→      existing.teacherId = teacherId;
   524→      return { ok: true, id: existing.id };
   525→    }
   526→    const id = genId();
   527→    cs.push({ id, classId, subjectId, teacherId, createdAt: nowIso() });
   528→    return { ok: true, id };
   529→  }, `assign: سپردن مضمون در صنف (${classId.slice(0, 10)})`);
   530→  return { ok: true };
   531→}
